class IoTDeviceIdentity {
public:
    void begin() {
        prefs.begin("iot-sec", true);
        String saved = prefs.getString("dev_id", "");
        prefs.end();
        if (saved.length() > 0) {
            saved.toCharArray(deviceId, sizeof(deviceId));
        } else {
            generateId();
            prefs.begin("iot-sec", false);
            prefs.putString("dev_id", String(deviceId));
            prefs.end();
        }
        const char* shortSuffix = deviceId + 4;
        snprintf(advertisedName, sizeof(advertisedName), "IoT-Lock-%s", shortSuffix);
        Serial.printf("[Identity] ID: %s | Nome: %s\n", deviceId, advertisedName);
    }
    const char* getDeviceId() { return deviceId; }
    const char* getAdvertisedName() { return advertisedName; }
private:
    Preferences prefs;
    char deviceId[16] = {0};
    char advertisedName[32] = {0};
    void generateId() {
        uint64_t chipId = ESP.getEfuseMac();
        snprintf(deviceId, sizeof(deviceId), "IoT_%05X", (unsigned int)(chipId & 0xFFFFF));
    }
};

void initStatusLeds() {
    pinMode(STATUS_LED_GREEN, OUTPUT);
    pinMode(STATUS_LED_WHITE, OUTPUT);
    pinMode(STATUS_LED_RED,   OUTPUT);
    digitalWrite(STATUS_LED_GREEN, LOW);
    digitalWrite(STATUS_LED_WHITE, LOW);
    digitalWrite(STATUS_LED_RED,   LOW);
}
void allLedsOff() {
    digitalWrite(STATUS_LED_GREEN, LOW);
    digitalWrite(STATUS_LED_WHITE, LOW);
    digitalWrite(STATUS_LED_RED,   LOW);
}
void setLedsApMode() {
    digitalWrite(STATUS_LED_GREEN, LOW);
    digitalWrite(STATUS_LED_WHITE, HIGH);
    digitalWrite(STATUS_LED_RED,   LOW);
}
void setLedsConnected() {
    digitalWrite(STATUS_LED_GREEN, HIGH);
    digitalWrite(STATUS_LED_WHITE, LOW);
    digitalWrite(STATUS_LED_RED,   LOW);
}
void setLedsError(bool blink = false) {
    static uint32_t lastBlink = 0;
    static bool blinkState = false;
    digitalWrite(STATUS_LED_GREEN, LOW);
    digitalWrite(STATUS_LED_WHITE, LOW);
    if (!blink) {
        digitalWrite(STATUS_LED_RED, HIGH);
    } else {
        uint32_t now = millis();
        if (now - lastBlink > 500) {
            lastBlink = now; blinkState = !blinkState;
            digitalWrite(STATUS_LED_RED, blinkState ? HIGH : LOW);
        }
    }
}
void updateStatusLeds(bool apMode = false) {
    if (apMode) { setLedsApMode(); return; }
    bool wifiOk = (WiFi.status() == WL_CONNECTED);
    bool mqttOk = mqttClient.connected();
    if (wifiOk && mqttOk) setLedsConnected();
    else                  setLedsError(wifiOk && !mqttOk);
}

void oledSplash(const char* l1, const char* l2 = nullptr, const char* l3 = nullptr) {
    if (!uiOledOk) return;
    oled.clearDisplay();
    oled.setTextColor(SSD1306_WHITE);
    oled.setTextSize(1);
    oled.setCursor(0, 0);
    oled.println(l1);
    if (l2) { oled.println(); oled.println(l2); }
    if (l3) { oled.println(l3); }

}

void uiFlash(const String& msg, uint32_t ms = 1500) {
    uiTransient   = msg;
    uiTransientMs = millis() + ms;
}

const char* mqttStateText(UiMqttState s) {
    switch (s) {
        case UI_MQTT_CONNECTED:  return "CONECTADO";
        case UI_MQTT_CONNECTING: return "CONECTANDO";
        case UI_MQTT_FAILED:     return "FALHA";
        case UI_AP_MODE:         return "AP";
        default:                 return "INIT";
    }
}

void displayDraw() {
    if (!uiOledOk) return;
    oled.clearDisplay();
    oled.setTextColor(SSD1306_WHITE);
    oled.setTextSize(1);

    // -------- Modo AP --------
    if (uiMqttState == UI_AP_MODE) {
        oled.setCursor(0, 0);
        oled.println("=== MODO CONFIG ===");
        oled.print("ID: ");
        oled.println(identity.getDeviceId());
        oled.drawFastHLine(0, 20, 128, SSD1306_WHITE);

        oled.setCursor(0, 24);
        oled.println("WiFi p/ configurar:");
        oled.println(AP_SSID);
        oled.println();
        oled.print("http://");
        oled.println(WiFi.softAPIP());
        oled.display();
        return;
    }

    // -------- Tela normal --------
    oled.setCursor(0, 0);
    oled.println("FECHADURA IOT NFC");
    oled.print("ID: ");
    oled.println(identity.getDeviceId());

    oled.drawFastHLine(0, 18, 128, SSD1306_WHITE);

    oled.setCursor(0, 22);
    oled.print("WiFi: ");
    if (WiFi.status() == WL_CONNECTED) {
        oled.print("OK ");
        oled.print(WiFi.RSSI());
        oled.println("dBm");
    } else {
        oled.println("DESCONECT");
    }

    oled.print("MQTT: ");
    oled.println(mqttStateText(uiMqttState));

    oled.print("IP: ");
    if (WiFi.status() == WL_CONNECTED) oled.println(WiFi.localIP());
    else                               oled.println("---");

    oled.drawFastHLine(0, 50, 128, SSD1306_WHITE);

    oled.setCursor(0, 54);
    oled.print(lockState == UNLOCKED ? "TRAVA:ABERTA " : "TRAVA:FECHADA");
    if (lockdown) oled.print(" LD");

    if (uiTransient.length() && millis() < uiTransientMs) {
        oled.fillRect(0, 54, 128, 10, SSD1306_BLACK);
        oled.setCursor(0, 54);
        oled.print(uiTransient);
    } else if (millis() >= uiTransientMs) {
        uiTransient = "";
    }

    oled.display();
}

void displayTick() {
    if (!uiOledOk) return;
    if (millis() - uiLastDraw < 400) return;
    uiLastDraw = millis();
    displayDraw();
}

String uidToHex(const uint8_t* uid, uint8_t len) {
    String s;
    for (uint8_t i = 0; i < len; i++) {
        if (i) s += ":";
        char b[3]; snprintf(b, sizeof(b), "%02X", uid[i]);
        s += b;
    }
    return s;
}
uint8_t parseUidHex(const String& h, uint8_t* out, uint8_t maxLen) {
    uint8_t len = 0; int start = 0;
    while (start < (int)h.length() && len < maxLen) {
        int colon = h.indexOf(':', start);
        String part = (colon < 0) ? h.substring(start) : h.substring(start, colon);
        out[len++] = (uint8_t)strtol(part.c_str(), nullptr, 16);
        if (colon < 0) break;
        start = colon + 1;
    }
    return len;
}

void saveCards() {
    JsonDocument doc;
    JsonArray arr = doc.to<JsonArray>();
    for (auto& kv : cardTable) {
        const CardEntry& c = kv.second;
        JsonObject o = arr.add<JsonObject>();
        o["uid"]     = kv.first;
        o["name"]    = c.name;
        o["en"]      = c.enabled;
        o["master"]  = c.isMaster;
        o["duress"]  = c.isDuress;
        o["guest"]   = c.isGuest;
        o["vf"]      = c.validFrom;
        o["vu"]      = c.validUntil;
        o["max"]     = c.maxUses;
        o["uses"]    = c.usesCount;
        o["wd"]      = c.weekdayMask;
        o["ts"]      = c.timeStartMin;
        o["te"]      = c.timeEndMin;
        o["cool"]    = c.cooldownMs;
    }
    prefs.begin("lock-cards", false);
    String out; serializeJson(doc, out);
    prefs.putString("data", out);
    prefs.end();
}
void loadCards() {
    prefs.begin("lock-cards", true);
    String in = prefs.getString("data", "");
    prefs.end();
    if (!in.length()) return;
    JsonDocument doc;
    if (deserializeJson(doc, in)) return;
    for (JsonObject o : doc.as<JsonArray>()) {
        CardEntry c;
        c.uidLen  = parseUidHex(String((const char*)(o["uid"] | "")), c.uid, UID_MAX_LEN);
        if (!c.uidLen) continue;
        c.name    = String((const char*)(o["name"] | ""));
        c.enabled = o["en"]     | true;
        c.isMaster= o["master"] | false;
        c.isDuress= o["duress"] | false;
        c.isGuest = o["guest"]  | false;
        c.validFrom  = o["vf"] | (uint32_t)0;
        c.validUntil = o["vu"] | (uint32_t)0;
        c.maxUses    = o["max"]| (uint16_t)0;
        c.usesCount  = o["uses"]|(uint16_t)0;
        c.weekdayMask= o["wd"] | (uint8_t)0;
        c.timeStartMin = o["ts"] | (uint16_t)0;
        c.timeEndMin   = o["te"] | (uint16_t)0;
        c.cooldownMs   = o["cool"] | (uint32_t)0;
        cardTable[String((const char*)(o["uid"] | ""))] = c;
    }
    Serial.printf("[LOCK] %d cartões carregados\n", (int)cardTable.size());
}

void saveHistory() {
    JsonDocument doc;
    JsonArray arr = doc.to<JsonArray>();
    for (const EventEntry& e : history) {
        JsonObject o = arr.add<JsonObject>();
        o["ts"]     = e.ts;
        o["uid"]    = e.uid;
        o["name"]   = e.name;
        o["result"] = e.result;
        o["source"] = e.source;
    }
    prefs.begin("lock-hist", false);
    String out; serializeJson(doc, out);
    prefs.putString("data", out);
    prefs.end();
}
void loadHistory() {
    prefs.begin("lock-hist", true);
    String in = prefs.getString("data", "");
    prefs.end();
    if (!in.length()) return;
    JsonDocument doc;
    if (deserializeJson(doc, in)) return;
    for (JsonObject o : doc.as<JsonArray>()) {
        EventEntry e;
        e.ts     = o["ts"] | (uint32_t)0;
        e.uid    = String((const char*)(o["uid"] | ""));
        e.name   = String((const char*)(o["name"] | ""));
        e.result = String((const char*)(o["result"] | ""));
        e.source = String((const char*)(o["source"] | ""));
        history.push_back(e);
    }
}

bool publishJson(JsonDocument& doc, const char* topic) {
    if (!mqttClient.connected()) return false;
    String out;
    if (serializeJson(doc, out) == 0) return false;
    bool ok = mqttClient.publish(topic, out.c_str(), 0);
    Serial.printf("[MQTT] %s %s (%u bytes)\n", ok ? "OK" : "FAIL", topic, (unsigned)out.length());
    return ok;
}
bool sendCommandResponse(long cmdId, bool success, const char* message,
                         JsonVariant result) {
    if (!mqttClient.connected()) return false;
    JsonDocument doc;
    doc["type"]       = "command_response";
    doc["command_id"] = cmdId;
    doc["success"]    = success;
    doc["message"]    = message;
    doc["timestamp"]  = millis();
    if (!result.isNull()) doc["result"] = result;
    return publishJson(doc, responseTopic);
}
bool sendTelemetry() {
    if (!mqttClient.connected()) return false;
    JsonDocument doc;
    doc["type"]         = "telemetry";
    doc["device_id"]    = identity.getDeviceId();
    doc["uptime"]       = uptimeSeconds;
    doc["lock_state"]   = (lockState == UNLOCKED) ? "unlocked" : "locked";
    doc["door_open"]    = doorOpen;
    doc["lockdown"]     = lockdown;
    doc["alarm"]        = alarmActive;
    doc["cards"]        = (int)cardTable.size();
    doc["grants"]       = grantCount;
    doc["denies"]       = denyCount;
    doc["unlocks"]      = unlockCount;
    doc["failed"]       = failedAttempts;
    doc["time_synced"]  = timeSynced;
    doc["free_heap"]    = ESP.getFreeHeap();
    doc["commands"]     = totalCommands;
    return publishJson(doc, statTopic);
}
bool sendHeartbeat() {
    if (!mqttClient.connected()) return false;
    JsonDocument doc;
    doc["esp_id"]      = identity.getDeviceId();
    doc["rssi"]        = WiFi.RSSI();
    doc["uptime"]      = uptimeSeconds;
    doc["free_heap"]   = ESP.getFreeHeap();
    doc["ip"]          = WiFi.localIP().toString();
    doc["lock_state"]  = (lockState == UNLOCKED) ? "unlocked" : "locked";
    doc["heartbeat"]   = true;
    return publishJson(doc, heartbeatTopic);
}
bool sendAlarm(const char* type, const char* value) {
    if (!mqttClient.connected()) return false;
    JsonDocument doc;
    doc["type"]       = "alarm";
    doc["alarm_type"] = type;
    doc["value"]      = value;
    doc["device_id"]  = identity.getDeviceId();
    doc["ts"]         = timeSynced ? (uint32_t)time(nullptr) : (millis() / 1000);
    return publishJson(doc, alarmTopic);
}
void pushEvent(const String& uid, const String& name,
               const String& result, const String& source) {
    EventEntry e;
    e.ts     = timeSynced ? (uint32_t)time(nullptr) : (millis() / 1000);
    e.uid    = uid; e.name = name; e.result = result; e.source = source;
    history.push_back(e);
    while ((int)history.size() > MAX_HISTORY) history.erase(history.begin());
    saveHistory();
    if (mqttClient.connected()) {
        JsonDocument doc;
        doc["type"]      = "access_event";
        doc["ts"]        = e.ts;
        doc["uid"]       = e.uid;
        doc["name"]      = e.name;
        doc["result"]    = e.result;
        doc["source"]    = e.source;
        doc["device_id"] = identity.getDeviceId();
        String out; serializeJson(doc, out);
        mqttClient.publish(eventsTopic, out.c_str(), 0);
    }
}

// NTP

void ntpSync() {
    configTime(0, 0, "pool.ntp.org", "time.nist.gov");
    time_t now = 0; int tries = 0;
    while (now < 1600000000 && tries < 20) {
        delay(500);
        now = time(nullptr);
        tries++;
    }
    if (now >= 1600000000) {
        timeSynced = true;
        Serial.printf("[NTP] Sincronizado: %s", ctime(&now));
    } else {
        Serial.println("[NTP] Falha — usando tempo relativo");
    }
}

void unlockDoor(uint32_t ttlMs, bool oneShot) {
    lockSet(true);
    lockState    = UNLOCKED;
    oneShotUnlock = oneShot;
    awaitingDoorCycle = oneShot;
    unlockUntil  = (oneShot || ttlMs == 0) ? 0 : (millis() + ttlMs);
    unlockCount++;
    Serial.printf("[LOCK] UNLOCK (ttl=%ums oneShot=%d)\n", (unsigned)ttlMs, oneShot);
    uiFlash("DESTRAVADA", 1200);
    displayDraw();
}
void lockDoor(const char* reason = "") {
    lockSet(false);
    lockState = LOCKED;
    unlockUntil = 0;
    oneShotUnlock = false;
    awaitingDoorCycle = false;
    Serial.printf("[LOCK] LOCK (%s)\n", reason);
    uiFlash("TRANCADA", 1000);
    displayDraw();
}
void updateDoorSensorAndAutoLock() {
    bool nowOpen = (digitalRead(DOOR_SENSOR_PIN) == LOW);
    if (nowOpen != doorOpen) {
        doorOpen = nowOpen;
        if (doorOpen) {
            doorOpenedAt = millis();
            doorAjarAlerted = false;
        } else {
            doorClosedAt = millis();
        }
    }

    if (doorOpen && !doorAjarAlerted && doorOpenedAt &&
        (millis() - doorOpenedAt > DOOR_AJAR_ALERT_MS)) {
        doorAjarAlerted = true;
        sendAlarm("door_ajar", "Door left open");
        pushEvent("", "", "alarm:door_ajar", "system");
    }

    if (!oneShotUnlock && lockState == UNLOCKED && unlockUntil && millis() >= unlockUntil) {
        lockDoor("timeout");
    }

    if (lockState == UNLOCKED && awaitingDoorCycle && doorClosedAt && !doorOpen &&
        (millis() - doorClosedAt > DOOR_RELOCK_DELAY_MS)) {
        lockDoor("door-closed");
    }
}

// ============================================================
// VALIDAÇÃO DE CARTÃO
// ============================================================
bool scheduleAllows(const CardEntry& c, time_t now) {
    if (c.weekdayMask == 0 && c.timeStartMin == 0 && c.timeEndMin == 0) return true;
    struct tm tmInfo; localtime_r(&now, &tmInfo);
    if (c.weekdayMask != 0 && !(c.weekdayMask & (1 << tmInfo.tm_wday))) return false;
    if (c.timeStartMin || c.timeEndMin) {
        uint16_t cur = tmInfo.tm_hour * 60 + tmInfo.tm_min;
        if (c.timeStartMin <= c.timeEndMin) {
            if (cur < c.timeStartMin || cur > c.timeEndMin) return false;
        } else {
            if (cur < c.timeStartMin && cur > c.timeEndMin) return false;
        }
    }
    return true;
}
bool validateCard(const CardEntry& c, const String& uidHex, String& reason) {
    if (!c.enabled)      { reason = "disabled";       return false; }
    if (!c.isMaster) {
        if (lockdown) { reason = "lockdown"; return false; }
        time_t now = time(nullptr);
        if (c.validFrom && timeSynced && (uint32_t)now < c.validFrom) {
            reason = "not_yet_valid"; return false;
        }
        if (c.validUntil && timeSynced && (uint32_t)now > c.validUntil) {
            reason = "expired"; return false;
        }
        if (c.maxUses && c.usesCount >= c.maxUses) {
            reason = "uses_exhausted"; return false;
        }
        if (timeSynced && !scheduleAllows(c, now)) {
            reason = "outside_schedule"; return false;
        }
    }
    if (c.cooldownMs && c.lastUsedMillis &&
        (millis() - c.lastUsedMillis < c.cooldownMs)) {
        reason = "cooldown"; return false;
    }
    return true;
}

void startEnroll(EnrollMode mode, const String& name,
                 uint32_t validUntilEpoch, uint16_t maxUses) {
    enrollMode = mode;
    enrollStart = millis();
    enrollPendingName = name;
    enrollPendingValidUntil = validUntilEpoch;
    enrollPendingMaxUses = maxUses;
    buzzEnroll();
    uiFlash("Aproxime cartao", enrollTimeoutMs);
    Serial.printf("[ENROLL] Modo %d iniciado\n", (int)mode);
}
void cancelEnroll() {
    enrollMode = ENR_NONE;
    Serial.println("[ENROLL] cancelado");
}

void handleCardRead(const uint8_t* uid, uint8_t uidLen) {
    String uidHex = uidToHex(uid, uidLen);
    Serial.printf("[NFC] Cartão: %s\n", uidHex.c_str());

    if (enrollMode != ENR_NONE) {
        if (cardTable.count(uidHex)) {
            Serial.println("[ENROLL] UID já cadastrado");
            buzzDenied();
            uiFlash("JA CADASTRADO", 1500);
            return;
        }
        CardEntry c;
        c.uidLen = uidLen;
        memcpy(c.uid, uid, uidLen);
        c.name = enrollPendingName.length() ? enrollPendingName
                                            : ("card_" + uidHex.substring(0,5));
        c.enabled = true;

        switch (enrollMode) {
            case ENR_MASTER: c.isMaster = true;  break;
            case ENR_DURESS: c.isDuress = true;  break;
            case ENR_GUEST:
                c.isGuest = true;
                c.validUntil = enrollPendingValidUntil;
                c.maxUses    = enrollPendingMaxUses;
                break;
            default: break;
        }
        cardTable[uidHex] = c;
        saveCards();
        buzzGranted();
        uiFlash("CADASTRADO", 1500);
        pushEvent(uidHex, c.name, "enrolled", "nfc");
        Serial.printf("[ENROLL] Cadastrado: %s (%s)\n", c.name.c_str(), uidHex.c_str());
        cancelEnroll();
        return;
    }

    auto it = cardTable.find(uidHex);
    if (it == cardTable.end()) {
        buzzDenied();
        denyCount++;
        failedAttempts++;
        if (!failedWindowStart) failedWindowStart = millis();
        uiFlash("NAO CADASTRADO", 1500);
        pushEvent(uidHex, "", "denied:unknown_card", "nfc");
        if (failedAttempts >= MAX_FAILED_ATTEMPTS) {
            alarmActive = true;
            sendAlarm("too_many_attempts", uidHex.c_str());
            pushEvent(uidHex, "", "alarm:too_many_attempts", "system");
            failedAttempts = 0;
        }
        return;
    }

    CardEntry& c = it->second;
    String reason;
    if (!validateCard(c, uidHex, reason)) {
        buzzDenied();
        denyCount++;
        uiFlash("NEGADO: " + reason, 1500);
        pushEvent(uidHex, c.name, "denied:" + reason, "nfc");
        Serial.printf("[NFC] Negado: %s (%s)\n", uidHex.c_str(), reason.c_str());
        if (reason == "disabled" || reason == "lockdown") {
            failedAttempts++;
            if (failedAttempts >= MAX_FAILED_ATTEMPTS) {
                alarmActive = true;
                sendAlarm("too_many_attempts", uidHex.c_str());
                failedAttempts = 0;
            }
        }
        return;
    }

    buzzGranted();
    grantCount++;
    failedAttempts = 0;
    c.usesCount++;
    c.lastUsedMillis = millis();
    if (timeSynced) c.lastUsedEpoch = (uint32_t)time(nullptr);
    saveCards();

    if (c.isDuress) {
        sendAlarm("duress", uidHex.c_str());
        pushEvent(uidHex, c.name, "granted:duress", "nfc");
    } else {
        pushEvent(uidHex, c.name, "granted", "nfc");
    }

    uiFlash("OK: " + c.name.substring(0, 14), 1500);
    unlockDoor(DEFAULT_UNLOCK_MS, false);
    Serial.printf("[NFC] Autorizado: %s (%s)\n", c.name.c_str(), uidHex.c_str());
}

void updateNFC() {
    uint8_t uid[UID_MAX_LEN];
    uint8_t uidLen = 0;
    if (nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, uid, &uidLen, 80)) {
        handleCardRead(uid, uidLen);
        delay(300);
    }
    if (enrollMode != ENR_NONE && (millis() - enrollStart > enrollTimeoutMs)) {
        Serial.println("[ENROLL] Timeout");
        uiFlash("TIMEOUT", 1500);
        cancelEnroll();
    }
}

JsonDocument fn_card_register_start(JsonArray args) {

JsonDocument fn_card_remove(JsonArray args) {
    const char* key = args[0] | "";
    if (!*key) return makeResponse(false, "missing uid or name");
    String k(key);
    if (cardTable.count(k)) {
        cardTable.erase(k);
        saveCards();
        return makeResponse(true, "removed by uid");
    }
    for (auto it = cardTable.begin(); it != cardTable.end(); ++it) {
        if (it->second.name == k) {
            cardTable.erase(it);
            saveCards();
            return makeResponse(true, "removed by name");
        }
    }
    return makeResponse(false, "not found");
}

void processCommand(const char* jsonStr, size_t len) {
    Serial.printf("[CMD] %.*s\n", (int)len, jsonStr);
    JsonDocument doc;
    if (deserializeJson(doc, jsonStr, len)) {
        Serial.println("[CMD] JSON inválido");
        return;
    }
    totalCommands++;
    const char* fnName = doc["fn"] | "";
    long cmdId         = doc["command_id"] | 0;
    JsonArray args     = doc["args"].as<JsonArray>();

    auto it = functionRegistry.find(String(fnName));
    if (it == functionRegistry.end()) {
        sendCommandResponse(cmdId, false, "unknown function");
        return;
    }
    JsonDocument result = it->second(args);
    bool success = result["success"] | false;
    const char* msg = result["message"] | "";
    JsonVariant data = result["result"];
    sendCommandResponse(cmdId, success, msg, data);
}

void mqttCallback(char* topic, byte* payload, unsigned int len) {
    char* buf = (char*)malloc(len + 1);
    if (!buf) return;
    memcpy(buf, payload, len); buf[len] = '\0';
    if (strcmp(topic, cmdTopic) == 0) processCommand(buf, len);
    free(buf);
}

void reconnectMQTT() {
    while (!mqttClient.connected()) {
        Serial.print("Conectando MQTT/TLS...");
        uiMqttState = UI_MQTT_CONNECTING;
        displayDraw();

        const char* id = identity.getDeviceId();
        if (mqttClient.connect(id, DEFAULT_MQTT_USER, DEFAULT_MQTT_PASS)) {
            Serial.println(" OK");
            uiMqttState = UI_MQTT_CONNECTED;
            mqttClient.subscribe(cmdTopic);
            JsonDocument doc;
            doc["type"] = "online";
            doc["device_id"] = id;
            publishJson(doc, statTopic);
            displayDraw();
            return;
        } else {
            Serial.printf(" falhou (rc=%d). retry em %ds\n",
                          mqttClient.state(), MQTT_RECONNECT_DELAY/1000);
            uiMqttState = UI_MQTT_FAILED;
            displayDraw();
            delay(MQTT_RECONNECT_DELAY);
        }
        updateStatusLeds(false);
    }
}

void handleRoot() {
    String html = FPSTR(INDEX_HTML);
    html.replace("%ESPID%", identity.getDeviceId());
    server.send(200, "text/html", html);
}
void handleScan() {
    int n = WiFi.scanComplete();
    if (n == -2) { WiFi.scanNetworks(true, true);
        server.send(200, "application/json", "{\"redes\":[]}"); return; }
    if (n < 0) { server.send(200, "application/json", "{\"redes\":[]}"); return; }
    JsonDocument doc; JsonArray arr = doc["redes"].to<JsonArray>();
    for (int i = 0; i < n; i++) {
        JsonObject o = arr.add<JsonObject>();
        o["ssid"] = WiFi.SSID(i);
        o["rssi"] = WiFi.RSSI(i);
        o["seg"]  = (WiFi.encryptionType(i) != WIFI_AUTH_OPEN);
    }
    String out; serializeJson(doc, out);
    WiFi.scanDelete();
    server.send(200, "application/json", out);
}
void handleSave() {
    String ssid = server.arg("ssid");
    String pass = server.arg("password");
    if (!ssid.length()) {
        server.send(400, "text/html", "<h1>SSID obrigatorio</h1>");
        return;
    }
    Preferences p; p.begin("wifi-config", false);
    p.putString("ssid", ssid); p.putString("password", pass); p.end();
    server.send(200, "text/html",
        "<!DOCTYPE html><html><head><meta charset='UTF-8'><title>OK</title></head>"
        "<body style='font-family:sans-serif;background:#0b0313;color:#e9defc;text-align:center;padding:60px'>"
        "<h2>Configuracao salva</h2><p>Reiniciando...</p></body></html>");
    delay(1500);
    ESP.restart();
}
void handleNotFound() {
    server.sendHeader("Location", "/", true);
    server.send(302, "text/plain", "");
}
void startAPMode() {
    Serial.println("[AP] Iniciando Access Point");
    WiFi.mode(WIFI_AP);
    WiFi.softAP(AP_SSID, AP_PASSWORD);
    IPAddress ip = WiFi.softAPIP();
    Serial.printf("[AP] IP: %s\n", ip.toString().c_str());

    setLedsApMode();
    uiMqttState = UI_AP_MODE;
    displayDraw();

    dnsServer.start(53, "*", ip);
    server.on("/", handleRoot);
    server.on("/scan", handleScan);
    server.on("/save", HTTP_POST, handleSave);
    server.onNotFound(handleNotFound);
    server.begin();
    Serial.println("[AP] Servidor HTTP iniciado");

    while (true) {
        dnsServer.processNextRequest();
        server.handleClient();
        displayTick();
        static unsigned long lb = 0;
        if (millis() - lb > 500) {
            lb = millis();
            static bool st = false; st = !st;
            digitalWrite(STATUS_LED_WHITE, st ? HIGH : LOW);
        }
        delay(10);
    }
}

void setup() {
    Serial.begin(115200);
    delay(500);
    Serial.println("\n================================================");
    Serial.println("  FECHADURA IoT NFC (PN532 + OLED + MQTT/TLS)");
    Serial.println("  PN532: SDA=26 SCL=27 (Wire1)");
    Serial.println("  OLED : SDA=21 SCL=22 (Wire)");
    Serial.println("================================================\n");

    // -------- 1) GPIOs --------
    initStatusLeds();
    allLedsOff();
    initLockHardware();

    // -------- 2) OLED primeiro --------
    Wire.begin(OLED_SDA, OLED_SCL);
    if (!oled.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR)) {
        Serial.println("[OLED] NAO detectado em 0x3C");
        uiOledOk = false;
    } else {
        uiOledOk = true;
        oled.clearDisplay();
        oled.setTextColor(SSD1306_WHITE);
        oled.setTextSize(1);
        oled.setCursor(0, 0);
        oled.println("FECHADURA IOT");
        oled.println();
        oled.println("Inicializando...");
        oled.display();
        Serial.println("[OLED] OK");
    }

    // -------- 3) Identidade --------
    identity.begin();
    const char* deviceId = identity.getDeviceId();

    if (uiOledOk) {
        oled.clearDisplay();
        oled.setCursor(0, 0);
        oled.println("FECHADURA IOT NFC");
        oled.println();
        oled.print("ID: "); oled.println(deviceId);
        oled.println();
        oled.println("Iniciando HW...");
        oled.display();
    }

    // -------- 4) Tópicos MQTT --------
    snprintf(cmdTopic,       sizeof(cmdTopic),       "iot/%s/command",   deviceId);
    snprintf(statTopic,      sizeof(statTopic),      "iot/%s/status",    deviceId);
    snprintf(alarmTopic,     sizeof(alarmTopic),     "iot/%s/alarm",     deviceId);
    snprintf(heartbeatTopic, sizeof(heartbeatTopic), "iot/%s/heartbeat", deviceId);
    snprintf(responseTopic,  sizeof(responseTopic),  "iot/%s/response",  deviceId);
    snprintf(eventsTopic,    sizeof(eventsTopic),    "iot/%s/events",    deviceId);

    // -------- 5) PN532 em Wire1 --------
    Wire1.begin(PN532_SDA, PN532_SCL);
    nfc.begin();
    uint32_t ver = nfc.getFirmwareVersion();
    if (!ver) {
        Serial.println("[NFC] PN532 NAO encontrado em Wire1");
        uiFlash("NFC OFFLINE", 3000);
    } else {
        Serial.printf("[NFC] PN532 fw %u.%u\n",
                      (ver >> 16) & 0xFF, (ver >> 8) & 0xFF);
        nfc.SAMConfig();
    }

    // -------- 6) Persistência --------
    loadCards();
    loadHistory();

    // -------- 7) WiFi + TLS --------
    WiFi.setTxPower(MAX_TX_POWER_WIFI);

#ifdef USE_MQTT_TLS_CA
    secureClient.setCACert(MQTT_ROOT_CA);
    Serial.println("[MQTT] TLS com CA");
#else
    secureClient.setInsecure();
    Serial.println("[MQTT] TLS sem verificacao");
#endif
    secureClient.setTimeout(MQTT_SOCKET_TIMEOUT);

    Preferences p; p.begin("wifi-config", true);
    String ssid = p.getString("ssid", "");
    String pass = p.getString("password", "");
    p.end();

    bool connected = false;
    if (ssid.length()) {
        Serial.printf("[WiFi] Conectando a '%s'\n", ssid.c_str());
        if (uiOledOk) {
            oled.clearDisplay();
            oled.setCursor(0, 0);
            oled.println("Conectando WiFi");
            oled.println();
            oled.println(ssid);
            oled.display();
        }
        WiFi.mode(WIFI_STA);
        WiFi.begin(ssid.c_str(), pass.c_str());
        int a = 0;
        while (WiFi.status() != WL_CONNECTED && a < 30) {
            delay(500);
            Serial.print(".");
            setLedsError(true);
            a++;
        }
        if (WiFi.status() == WL_CONNECTED) {
            connected = true;
            Serial.printf("\n[WiFi] IP: %s\n", WiFi.localIP().toString().c_str());
            if (uiOledOk) {
                oled.clearDisplay();
                oled.setCursor(0, 0);
                oled.println("WiFi OK");
                oled.println();
                oled.println(WiFi.localIP());
                oled.display();
                delay(700);
            }
        } else {
            Serial.println("\n[WiFi] Falha");
            if (uiOledOk) {
                oled.clearDisplay();
                oled.setCursor(0, 0);
                oled.println("WiFi FALHOU");
                oled.println();
                oled.println("Abrindo portal...");
                oled.display();
                delay(700);
            }
        }
    } else {
        if (uiOledOk) {
            oled.clearDisplay();
            oled.setCursor(0, 0);
            oled.println("Sem WiFi salvo");
            oled.println();
            oled.println("Abrindo portal...");
            oled.display();
            delay(700);
        }
    }

    // Sem WiFi → AP mode (nunca retorna)
    if (!connected) startAPMode();

    // -------- 8) NTP --------
    ntpSync();

    // -------- 9) MQTT --------
    setLedsError(true);
    registerFunctions();
    mqttClient.setServer(DEFAULT_MQTT_BROKER, DEFAULT_MQTT_PORT);
    mqttClient.setCallback(mqttCallback);
    mqttClient.setKeepAlive(MY_KEEPALIVE);

    uiMqttState = UI_MQTT_CONNECTING;
    displayDraw();

    reconnectMQTT();
    setLedsConnected();

    uiFlash("Pronto", 2000);
    displayDraw();
    Serial.println("Fechadura NFC pronta.");
}

// ============================================================
// LOOP
// ============================================================
void loop() {
    if (WiFi.status() != WL_CONNECTED) {
        Serial.println("[WiFi] Reconectando...");
        uiMqttState = UI_MQTT_FAILED;
        WiFi.reconnect();
        displayTick();
        delay(WIFI_RECONNECT_DELAY);
        updateStatusLeds(false);
        return;
    }

    if (!mqttClient.connected()) {
        uiMqttState = UI_MQTT_FAILED;
        reconnectMQTT();
    }
    mqttClient.loop();

    updateNFC();
    updateDoorSensorAndAutoLock();

    uint32_t now = millis();
    if (mqttClient.connected() && (now - lastTelemetry >= TELEMETRY_INTERVAL)) {
        lastTelemetry = now;
        sendTelemetry();
    }
    if (mqttClient.connected() && (now - lastHeartbeat >= HEARTBEAT_INTERVAL)) {
        lastHeartbeat = now;
        sendHeartbeat();
    }
    if (now - lastUptimeUpdate >= 1000) {
        lastUptimeUpdate = now;
        uptimeSeconds++;
        if (failedWindowStart && (now - failedWindowStart > 3600000UL)) {
            failedAttempts = 0; failedWindowStart = 0;
        }
    }

    static uint32_t lastStatusUpdate = 0;
    if (now - lastStatusUpdate >= 2000) {
        lastStatusUpdate = now;
        updateStatusLeds(false);
    }

    displayTick();
    delay(10);
}
