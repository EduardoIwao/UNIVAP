const
   PONTOS_VITORIA = 3
   PONTOS_EMPATE = 1
   PONTOS_DERROTA = 0
   MAX_EQUIPES = 10
   RES_VITORIA = 1     // codigos internos do resultado de uma partida
   RES_EMPATE = 2
   RES_DERROTA = 3

tipo
   // Registro que representa logicamente uma equipe
   tEquipe = registro
      codigo: inteiro
      nome: caractere
      capitao: caractere
      vitorias: inteiro
      empates: inteiro
      derrotas: inteiro
      pontosMarcados: inteiro
      pontosSofridos: inteiro
      saldo: inteiro
      pontuacao: inteiro
   fimregistro

var
   // vetor de registros (tamanho = MAX_EQUIPES)
   equipes: vetor[1..10] de tEquipe

   // matrizes de confrontos
   matJogos: vetor[1..10,1..10] de inteiro
   matPontosPro: vetor[1..10,1..10] de inteiro

   // estado geral do campeonato
   qtdEquipes: inteiro
   totalPartidas: inteiro
   opcao: inteiro

// ============================================================================
// SUB-ROTINAS
// ============================================================================

// Zera os contadores e as matrizes (chamado uma vez no inicio)
procedimento inicializarSistema
var
   i, j: inteiro
inicio
   qtdEquipes <- 0
   totalPartidas <- 0
   para i de 1 ate MAX_EQUIPES faca
      equipes[i].codigo <- 0
      equipes[i].nome <- ""
      equipes[i].capitao <- ""
      equipes[i].vitorias <- 0
      equipes[i].empates <- 0
      equipes[i].derrotas <- 0
      equipes[i].pontosMarcados <- 0
      equipes[i].pontosSofridos <- 0
      equipes[i].saldo <- 0
      equipes[i].pontuacao <- 0
      // repeticao aninhada: zera a linha i das matrizes
      para j de 1 ate MAX_EQUIPES faca
         matJogos[i,j] <- 0
         matPontosPro[i,j] <- 0
      fimpara
   fimpara
fimprocedimento

// RECURSIVIDADE: contagem regressiva para o inicio do campeonato.
// Caso-base: n <= 0. Chamada recursiva: contagemRegressiva(n - 1).
procedimento contagemRegressiva(n: inteiro)
inicio
   se n <= 0 entao
      escreval("")
      escreval("CAMPEONATO INICIADO!")
   senao
      escreva(n, "... ")
      contagemRegressiva(n - 1)
   fimse
fimprocedimento

// Escreve um texto e completa com espacos ate a largura informada
procedimento escreverPad(texto: caractere; largura: inteiro)
var
   i: inteiro
inicio
   escreva(texto)
   para i de compr(texto) + 1 ate largura faca
      escreva(" ")
   fimpara
fimprocedimento

// Um texto e valido se tiver ao menos um caractere diferente de espaco
funcao nomeValido(texto: caractere): logico
var
   i: inteiro
   achouLetra: logico
inicio
   achouLetra <- falso
   para i de 1 ate compr(texto) faca
      se copia(texto, i, 1) <> " " entao
         achouLetra <- verdadeiro
      fimse
   fimpara
   retorne achouLetra
fimfuncao

// Le um inteiro >= 0, repetindo ate o usuario digitar um valor valido
funcao lerNaoNegativo(mensagem: caractere): inteiro
var
   valor: inteiro
inicio
   repita
      escreva(mensagem)
      leia(valor)
      se valor < 0 entao
         escreval("Valor invalido! Informe um numero maior ou igual a zero.")
      fimse
   ate valor >= 0
   retorne valor
fimfuncao

// Funcao de busca: devolve o indice da equipe no vetor (0 se nao encontrada)
funcao localizarEquipe(codigo: inteiro): inteiro
var
   i, posicao: inteiro
inicio
   posicao <- 0
   i <- 1
   enquanto (i <= qtdEquipes) e (posicao = 0) faca
      se equipes[i].codigo = codigo entao
         posicao <- i
      fimse
      i <- i + 1
   fimenquanto
   retorne posicao
fimfuncao

// Funcao: saldo de pontos = marcados - sofridos
funcao calcularSaldo(marcados: inteiro; sofridos: inteiro): inteiro
inicio
   retorne marcados - sofridos
fimfuncao

// Funcao: pontuacao no campeonato a partir de vitorias, empates e derrotas
funcao calcularPontuacao(vitorias: inteiro; empates: inteiro; derrotas: inteiro): inteiro
inicio
   retorne vitorias * PONTOS_VITORIA + empates * PONTOS_EMPATE + derrotas * PONTOS_DERROTA
fimfuncao

// PASSAGEM POR REFERENCIA (var): res1 e res2 sao alterados aqui e o novo valor
// precisa voltar para quem chamou.
procedimento determinarResultado(pontos1: inteiro; pontos2: inteiro; var res1: inteiro; var res2: inteiro)
inicio
   se pontos1 > pontos2 entao
      res1 <- RES_VITORIA
      res2 <- RES_DERROTA
   senao
      se pontos2 > pontos1 entao
         res1 <- RES_DERROTA
         res2 <- RES_VITORIA
      senao
         res1 <- RES_EMPATE
         res2 <- RES_EMPATE
      fimse
   fimse
fimprocedimento

// Atualiza as estatisticas de UMA equipe (idx) apos uma partida
procedimento atualizarEquipe(idx: inteiro; marcados: inteiro; sofridos: inteiro; resultado: inteiro)
inicio
   equipes[idx].pontosMarcados <- equipes[idx].pontosMarcados + marcados
   equipes[idx].pontosSofridos <- equipes[idx].pontosSofridos + sofridos
   se resultado = RES_VITORIA entao
      equipes[idx].vitorias <- equipes[idx].vitorias + 1
   senao
      se resultado = RES_EMPATE entao
         equipes[idx].empates <- equipes[idx].empates + 1
      senao
         equipes[idx].derrotas <- equipes[idx].derrotas + 1
      fimse
   fimse
   equipes[idx].saldo <- calcularSaldo(equipes[idx].pontosMarcados, equipes[idx].pontosSofridos)
   equipes[idx].pontuacao <- calcularPontuacao(equipes[idx].vitorias, equipes[idx].empates, equipes[idx].derrotas)
fimprocedimento

// Cadastro de equipe com todas as validacoes exigidas
procedimento cadastrarEquipe
var
   codigo, posicao: inteiro
   nome, capitao: caractere
inicio
   se qtdEquipes >= MAX_EQUIPES entao
      escreval("Capacidade maxima atingida (", MAX_EQUIPES, " equipes). Cadastro nao permitido.")
   senao
      escreva("Codigo da equipe (inteiro positivo): ")
      leia(codigo)
      se codigo <= 0 entao
         escreval("Codigo invalido! Deve ser um numero inteiro maior que zero.")
      senao
         se localizarEquipe(codigo) <> 0 entao
            escreval("Codigo duplicado! Ja existe uma equipe com o codigo ", codigo, ".")
         senao
            escreva("Nome da equipe: ")
            leia(nome)
            se nao nomeValido(nome) entao
               escreval("Nome da equipe nao pode ser vazio!")
            senao
               escreva("Nome do capitao: ")
               leia(capitao)
               se nao nomeValido(capitao) entao
                  escreval("Nome do capitao nao pode ser vazio!")
               senao
                  qtdEquipes <- qtdEquipes + 1
                  posicao <- qtdEquipes
                  equipes[posicao].codigo <- codigo
                  equipes[posicao].nome <- nome
                  equipes[posicao].capitao <- capitao
                  // estatisticas inicializadas
                  equipes[posicao].vitorias <- 0
                  equipes[posicao].empates <- 0
                  equipes[posicao].derrotas <- 0
                  equipes[posicao].pontosMarcados <- 0
                  equipes[posicao].pontosSofridos <- 0
                  equipes[posicao].saldo <- 0
                  equipes[posicao].pontuacao <- 0
                  escreval("Equipe ", nome, " cadastrada com sucesso!")
               fimse
            fimse
         fimse
      fimse
   fimse
fimprocedimento

// Exibe todas as informacoes de uma equipe (idx = posicao no vetor)

   escreval("Codigo.............: ", equipes[idx].codigo)

   escreval("Derrotas...........: ", equipes[idx].derrotas)

   escreval("Saldo de pontos....: ", equipes[idx].saldo)
   escreval("Pontuacao..........: ", equipes[idx].pontuacao)
fimprocedimento

procedimento listarEquipes
var
   i: inteiro
inicio
   se qtdEquipes = 0 entao
      escreval("Nenhuma equipe cadastrada.")
   senao
      escreval("Cod  Equipe               Capitao")
      escreval("---------------------------------------------")
      para i de 1 ate qtdEquipes faca
         escreva(equipes[i].codigo:3, "  ")
         escreverPad(equipes[i].nome, 20)
         escreva(" ")
         escreverPad(equipes[i].capitao, 20)
         escreval("")
      fimpara
   fimse
fimprocedimento

// Registro de partida: valida, define o resultado, atualiza equipes e matrizes
procedimento registrarPartida
var
   cod1, cod2, posA, posB, pts1, pts2, res1, res2: inteiro
inicio
   se qtdEquipes < 2 entao
      escreval("E necessario ter ao menos 2 equipes cadastradas para registrar uma partida.")
   senao
      escreva("Codigo da primeira equipe: ")
      leia(cod1)
      escreva("Codigo da segunda equipe: ")
      leia(cod2)
      posA <- localizarEquipe(cod1)
      posB <- localizarEquipe(cod2)
      se (posA = 0) ou (posB = 0) entao
         escreval("Partida nao registrada: uma das equipes (ou ambas) nao existe.")
      senao
         se posA = posB entao
            escreval("Partida nao registrada: uma equipe nao pode jogar contra ela propria.")
         senao
            pts1 <- lerNaoNegativo("Pontuacao da primeira equipe: ")
            pts2 <- lerNaoNegativo("Pontuacao da segunda equipe: ")
            determinarResultado(pts1, pts2, res1, res2)
            atualizarEquipe(posA, pts1, pts2, res1)
            atualizarEquipe(posB, pts2, pts1, res2)
            // matrizes de confrontos
            matJogos[posA,posB] <- matJogos[posA,posB] + 1
            matJogos[posB,posA] <- matJogos[posB,posA] + 1
            matPontosPro[posA,posB] <- matPontosPro[posA,posB] + pts1
            matPontosPro[posB,posA] <- matPontosPro[posB,posA] + pts2
            totalPartidas <- totalPartidas + 1
            escreval("Partida registrada: ", equipes[posA].nome, " ", pts1, " x ", pts2, " ", equipes[posB].nome)
            se res1 = RES_VITORIA entao
               escreval("Vencedor: ", equipes[posA].nome)
            senao
               se res2 = RES_VITORIA entao
                  escreval("Vencedor: ", equipes[posB].nome)
               senao
                  escreval("Resultado: EMPATE")
               fimse
            fimse
         fimse
      fimse
   fimse
fimprocedimento

   leia(codigo)
   posicao <- localizarEquipe(codigo)
   se posicao = 0 entao
      escreval("Equipe nao localizada.")
   senao
      exibirEquipe(posicao)
   fimse
fimprocedimento

// Funcao: decide se a equipe 'b' (logo abaixo) deve subir acima da equipe 'a'.
// Criterios: 1) pontuacao 2) vitorias 3) saldo 4) pontos marcados
funcao deveTrocar(a: inteiro; b: inteiro): logico
var
   trocar: logico
inicio
   trocar <- falso
   se equipes[b].pontuacao > equipes[a].pontuacao entao
      trocar <- verdadeiro
   senao
      se equipes[b].pontuacao = equipes[a].pontuacao entao
         se equipes[b].vitorias > equipes[a].vitorias entao
            trocar <- verdadeiro
         senao
            se equipes[b].vitorias = equipes[a].vitorias entao
               se equipes[b].saldo > equipes[a].saldo entao
                  trocar <- verdadeiro
               senao
                  se equipes[b].saldo = equipes[a].saldo entao
                     se equipes[b].pontosMarcados > equipes[a].pontosMarcados entao
                        trocar <- verdadeiro
                     fimse
                  fimse
               fimse
            fimse
         fimse
      fimse
   fimse
   retorne trocar
fimfuncao

// Troca duas equipes de posicao (todos os campos do registro) e tambem as
// linhas e colunas correspondentes das matrizes
procedimento trocarEquipes(a: inteiro; b: inteiro)
var
   k, auxInt: inteiro
   auxTxt: caractere
inicio
   auxInt <- equipes[a].codigo
   equipes[a].codigo <- equipes[b].codigo
   equipes[b].codigo <- auxInt

   auxTxt <- equipes[a].nome
   equipes[a].nome <- equipes[b].nome
   equipes[b].nome <- auxTxt

   auxTxt <- equipes[a].capitao
   equipes[a].capitao <- equipes[b].capitao
   equipes[b].capitao <- auxTxt

   auxInt <- equipes[a].vitorias
   equipes[a].vitorias <- equipes[b].vitorias
   equipes[b].vitorias <- auxInt

   auxInt <- equipes[a].empates
   equipes[a].empates <- equipes[b].empates
   equipes[b].empates <- auxInt

   auxInt <- equipes[a].derrotas
   equipes[a].derrotas <- equipes[b].derrotas
   equipes[b].derrotas <- auxInt

   auxInt <- equipes[a].pontosMarcados
   equipes[a].pontosMarcados <- equipes[b].pontosMarcados
   equipes[b].pontosMarcados <- auxInt

   auxInt <- equipes[a].pontosSofridos
   equipes[a].pontosSofridos <- equipes[b].pontosSofridos
   equipes[b].pontosSofridos <- auxInt

   auxInt <- equipes[a].saldo
   equipes[a].saldo <- equipes[b].saldo
   equipes[b].saldo <- auxInt

   auxInt <- equipes[a].pontuacao
   equipes[a].pontuacao <- equipes[b].pontuacao
   equipes[b].pontuacao <- auxInt

   // troca as LINHAS a e b das matrizes
   para k de 1 ate qtdEquipes faca
      auxInt <- matJogos[a,k]
      matJogos[a,k] <- matJogos[b,k]
      matJogos[b,k] <- auxInt
      auxInt <- matPontosPro[a,k]
      matPontosPro[a,k] <- matPontosPro[b,k]
      matPontosPro[b,k] <- auxInt
   fimpara
   // troca as COLUNAS a e b das matrizes
   para k de 1 ate qtdEquipes faca
      auxInt <- matJogos[k,a]
      matJogos[k,a] <- matJogos[k,b]
      matJogos[k,b] <- auxInt
      auxInt <- matPontosPro[k,a]
      matPontosPro[k,a] <- matPontosPro[k,b]
      matPontosPro[k,b] <- auxInt
   fimpara
fimprocedimento

// ORDENACAO: Bubble Sort (a cada passada o pior colocado "afunda" para o fim)
procedimento ordenarClassificacao
var
   i, j: inteiro
inicio
   para i de 1 ate qtdEquipes - 1 faca
      para j de 1 ate qtdEquipes - i faca
         se deveTrocar(j, j + 1) entao
            trocarEquipes(j, j + 1)
         fimse
      fimpara
   fimpara
fimprocedimento

procedimento exibirClassificacao
var
   i: inteiro
inicio
   se qtdEquipes = 0 entao
      escreval("Nenhuma equipe cadastrada.")
   senao
      ordenarClassificacao
      escreval("=================== CLASSIFICACAO GERAL ===================")
      escreval("Pos Equipe                 P    V    E    D   PM   PS   SG")
      escreval("-----------------------------------------------------------")
      para i de 1 ate qtdEquipes faca
         escreva(i:3, " ")
         escreverPad(equipes[i].nome, 20)
         escreva(equipes[i].pontuacao:5, equipes[i].vitorias:5, equipes[i].empates:5, equipes[i].derrotas:5)
         escreva(equipes[i].pontosMarcados:5, equipes[i].pontosSofridos:5, equipes[i].saldo:5)
         escreval("")
      fimpara
      escreval("P=Pontos V=Vitorias E=Empates D=Derrotas PM=Pontos Marcados PS=Pontos Sofridos SG=Saldo")
   fimse
fimprocedimento

procedimento exibirEstatisticas
var
   i, somaVitorias, somaEmpates, somaMarcados: inteiro
   idxMaxVit, idxMaxPts, idxMaxSaldo, idxMaxMarc: inteiro
   mediaPartida: real
inicio
   escreval("============ ESTATISTICAS DO CAMPEONATO ============")
   escreval("Equipes cadastradas.......: ", qtdEquipes)
   se qtdEquipes = 0 entao
      escreval("Sem dados para calcular as demais estatisticas.")
   senao
      somaVitorias <- 0
      somaEmpates <- 0
      somaMarcados <- 0
      idxMaxVit <- 1
      idxMaxPts <- 1
      idxMaxSaldo <- 1
      idxMaxMarc <- 1
      para i de 1 ate qtdEquipes faca
         somaVitorias <- somaVitorias + equipes[i].vitorias
         somaEmpates <- somaEmpates + equipes[i].empates
         somaMarcados <- somaMarcados + equipes[i].pontosMarcados
         se equipes[i].vitorias > equipes[idxMaxVit].vitorias entao
            idxMaxVit <- i
         fimse
         se equipes[i].pontuacao > equipes[idxMaxPts].pontuacao entao
            idxMaxPts <- i
         fimse
         se equipes[i].saldo > equipes[idxMaxSaldo].saldo entao
            idxMaxSaldo <- i
         fimse
         se equipes[i].pontosMarcados > equipes[idxMaxMarc].pontosMarcados entao
            idxMaxMarc <- i
         fimse
      fimpara
      escreval("Partidas realizadas.......: ", totalPartidas)
      // cada partida decidida gera 1 vitoria; cada empate e contado nas 2 equipes
      escreval("Partidas com vitoria......: ", somaVitorias)
      escreval("Partidas empatadas........: ", somaEmpates div 2)
      se totalPartidas > 0 entao
         // soma dos pontos das duas equipes dividida pelo numero de partidas
         mediaPartida <- somaMarcados / totalPartidas
         escreval("Media de pontos marcados por partida (soma das duas equipes): ", mediaPartida:6:2)
         escreval("Media de pontos marcados por equipe por partida: ", mediaPartida / 2:6:2)
      senao
         escreval("Media de pontos por partida: sem partidas realizadas.")
      fimse
      escreval("Mais vitorias.............: ", equipes[idxMaxVit].nome, " (", equipes[idxMaxVit].vitorias, ")")
      escreval("Maior pontuacao...........: ", equipes[idxMaxPts].nome, " (", equipes[idxMaxPts].pontuacao, ")")
      escreval("Maior saldo...............: ", equipes[idxMaxSaldo].nome, " (", equipes[idxMaxSaldo].saldo, ")")
      escreval("Mais pontos marcados......: ", equipes[idxMaxMarc].nome, " (", equipes[idxMaxMarc].pontosMarcados, ")")
   fimse
fimprocedimento

// Mostra a matriz de confrontos (numero de partidas) e detalha uma equipe
procedimento consultarConfrontos
var
   i, j, codigo, posicao: inteiro
inicio
   se qtdEquipes < 2 entao
      escreval("E necessario ter ao menos 2 equipes cadastradas.")
   senao
      escreval("MATRIZ DE CONFRONTOS (numero de partidas entre as equipes)")
      escreva("Cod  ")
      para j de 1 ate qtdEquipes faca
         escreva(equipes[j].codigo:5)
      fimpara
      escreval("")
      para i de 1 ate qtdEquipes faca
         escreva(equipes[i].codigo:3, "  ")
         para j de 1 ate qtdEquipes faca
            se i = j entao
               escreva("    -")
            senao
               escreva(matJogos[i,j]:5)
            fimse
         fimpara
         escreval("")
      fimpara
      escreval("")
      escreva("Codigo da equipe para detalhar (0 = voltar): ")
      leia(codigo)
      se codigo <> 0 entao
         posicao <- localizarEquipe(codigo)
         se posicao = 0 entao
            escreval("Equipe nao localizada.")
         senao
            escreval("Confrontos de ", equipes[posicao].nome, ":")
            para j de 1 ate qtdEquipes faca
               se j <> posicao entao
                  se matJogos[posicao,j] > 0 entao
                     escreval("  vs ", equipes[j].nome, ": ", matJogos[posicao,j], " jogo(s) | marcou ", matPontosPro[posicao,j], " | sofreu ", matPontosPro[j,posicao])
                  senao
                     escreval("  vs ", equipes[j].nome, ": ainda nao se enfrentaram")
                  fimse
               fimse
            fimpara
         fimse
      fimse
   fimse
fimprocedimento
